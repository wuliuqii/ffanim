//! 装配件场景族：花括号 / 定界符 / 箭头 / 数字滚动（Typst MATH stretch 装配）。
//!
//! 对齐 manim `example_scenes/` 形态：单文件多场景 + argv 寻址。
//!
//! 运行（仓库根目录）：
//! `cargo run -p ffanim --release --example assemblies -- [场景名...]`（缺省全渲染，≙ `manim -a`）
//! `cargo run -p ffanim --example assemblies -- --json brace_demo`（打印调度表，agent 检查用）

use ffanim::kurbo::Shape as _;
use ffanim::kurbo::{Affine, Point, Rect};
use ffanim::{Anim, Font, Item, Theme, Timeline};

const FPS: u32 = 30;

/// 场景清单（argv 寻址，缺省全渲染）。
const SCENES: &[&str] = &["brace_demo", "delimiter_demo", "arrow_demo", "decimal_demo"];

fn build(name: &str, theme: &Theme) -> ffanim::Result<Timeline> {
    match name {
        "brace_demo" => brace_demo(theme, FPS),
        "delimiter_demo" => delimiter_demo(theme, FPS),
        "arrow_demo" => arrow_demo(theme, FPS),
        "decimal_demo" => decimal_demo(theme, FPS),
        other => Err(ffanim::Error::Other(format!(
            "未知场景 {other}，可用场景: {}",
            SCENES.join(", ")
        ))),
    }
}

fn main() -> ffanim::Result<()> {
    let mut args: Vec<String> = std::env::args().skip(1).collect();
    let json = args.first().is_some_and(|a| a == "--json");
    if json {
        args.remove(0);
    }
    let names: Vec<String> = if args.is_empty() {
        SCENES.iter().map(|s| (*s).to_string()).collect()
    } else {
        args
    };
    // 主题路径相对 crate 目录向上两级到仓库根（与 fourier.rs 同形态）
    let theme_path = concat!(env!("CARGO_MANIFEST_DIR"), "/../../themes/default.json");
    let theme = Theme::from_path(theme_path)?;
    let _ = std::fs::create_dir_all("out");
    for name in &names {
        let tl = build(name, &theme)?.compile();
        if json {
            println!(
                "{}",
                serde_json::json!({
                    "scene": name,
                    "fps": tl.fps,
                    "duration_frames": tl.duration_frames,
                    "duration_seconds": tl.duration_frames as f64 / f64::from(tl.fps),
                    "items": tl.items.len(),
                    "anims": tl.anims.iter().map(|a| ffanim::AnimKind::describe(&a.kind)).collect::<Vec<_>>(),
                })
            );
        } else {
            ffanim::render_to_mp4(&tl, format!("out/{name}.mp4"))?;
            println!("{name}: {} 帧 → out/{name}.mp4", tl.duration_frames);
        }
    }
    Ok(())
}

/// M9.12 验收场景（P1-8）：Brace 花括号 + 括注文本。
///
/// 标题淡入 → 横条描边生长 → 下方花括号淡入放大 + 括注“宽度”淡入 →
/// 右侧花括号 + 括注“高度”（manim `Brace` / `brace.get_text` 对应）。
/// 花括号为填充字形，出场用 fade_in（manim `GrowFromCenter` 对应）而非 create。
fn brace_demo(theme: &Theme, fps: u32) -> ffanim::Result<Timeline> {
    use ffanim::Direction;
    use ffanim::typst::TypstEngine;

    // 括注为中文：须 CJK 字形（find_cjk_font 优先，缺失时回退系统拉丁字体）
    let font = ffanim::typst::find_cjk_font()
        .and_then(|(path, idx)| ffanim::Font::from_index(&path, idx).ok())
        .map(Ok)
        .unwrap_or_else(|| Font::from_system(None::<&str>))?;
    // 花括号 = Typst MATH 装配（NewCMMath），与公式排版同引擎
    let mut engine = TypstEngine::new(8);
    let accent = theme.accent().unwrap();
    let fg = theme.foreground().unwrap();
    let (w, h) = (960u32, 540u32);
    let mut tl = Timeline::new(w, h, fps, theme.background().unwrap());

    // 1) 标题淡入
    let text_path = font.outline_text("Brace 括注", 56.0).unwrap();
    let bounds = text_path.bounding_box();
    let dx = f64::from(w) / 2.0 - (bounds.x0 + bounds.x1) / 2.0;
    let title = tl.add(
        Item::from_path(text_path)
            .with_fill(fg)
            .with_transform(Affine::translate((dx, f64::from(h) * 0.10))),
    );
    tl.play(Anim::fade_in(title).during(0.7));

    // 2) 中央横条描边生长（留未注册副本作 brace 目标：bbox 只看几何）
    let bar_item = Item::rect(Rect::new(220.0, 200.0, 740.0, 310.0)).with_fill(accent);
    let bar = tl.add(bar_item.clone());
    tl.play(Anim::create(bar).during(1.0));

    // 3) 下方花括号 + 括注“宽度”
    let (brace_d, label_d) =
        engine.brace_with_text(&bar_item, Direction::Down, 14.0, &font, "宽度", 30.0)?;
    let (bd, ld) = (tl.add(brace_d.with_fill(fg)), tl.add(label_d.with_fill(fg)));
    tl.wait(0.2);
    tl.together(
        [Anim::fade_in(bd).during(1.0)]
            .into_iter()
            .chain([Anim::fade_in(ld).during(0.9)]),
    );

    // 4) 右侧花括号 + 括注“高度”
    let (brace_r, label_r) =
        engine.brace_with_text(&bar_item, Direction::Right, 14.0, &font, "高度", 30.0)?;
    let (br, lr) = (
        tl.add(brace_r.with_fill(accent)),
        tl.add(label_r.with_fill(accent)),
    );
    tl.wait(0.4);
    tl.together(
        [Anim::fade_in(br).during(1.0)]
            .into_iter()
            .chain([Anim::fade_in(lr).during(0.9)]),
    );

    tl.wait(0.8);
    Ok(tl)
}

        .map(Ok)
        .unwrap_or_else(|| Font::from_system(None::<&str>))?;
    // 定界符 = Typst MATH 装配（NewCMMath GlyphAssembly），与公式排版同引擎
    let mut engine = TypstEngine::new(8);
    let accent = theme.accent().unwrap();
    let fg = theme.foreground().unwrap();

    let mut tl = Timeline::new(w, h, fps, theme.background().unwrap());

    let dx = f64::from(w) / 2.0 - (bounds.x0 + bounds.x1) / 2.0;

    let bar_item = Item::rect(Rect::new(280.0, 210.0, 680.0, 310.0)).with_fill(accent);
    let bar = tl.add(bar_item.clone());
    tl.play(Anim::create(bar).during(1.0));

    // 3) 四类定界符依次贴四侧 + 括注（交替前景/强调色；每类同窗淡入）
    let specs: [(Delimiter, Direction, &str); 4] = [
        (Delimiter::Brace, Direction::Down, "宽度"),
        (Delimiter::Bracket, Direction::Up, "上沿"),
        (Delimiter::Paren, Direction::Left, "左沿"),
        (Delimiter::DoubleBracket, Direction::Right, "右沿"),
    ];
    for (i, (kind, dir, label)) in specs.into_iter().enumerate() {
        let (delim, text) =
            engine.delimiter_with_text(&bar_item, kind, dir, 14.0, &font, label, 30.0)?;
        let color = if i % 2 == 0 { fg } else { accent };
        let (d, t) = (
            tl.add(delim.with_fill(color)),
            tl.add(text.with_fill(color)),
        );
        tl.wait(0.15);
        tl.together(
            [Anim::fade_in(d).during(0.6)]
                .into_iter()
                .chain([Anim::fade_in(t).during(0.5)]),
        );
    }

    tl.wait(0.8);
    Ok(tl)
}

/// arrow_demo：Typst 装配箭头验收示例（M9.14）。
///
/// 标题淡入 → 三支箭头（水平单头 / 斜向单头 / 水平双头）依次淡入 +
/// 中文括注（manim `Arrow`/`DoubleArrow` 对应；头高随跨度自适应并封顶，
/// 双头两端对称）。形状 = Typst MATH stretch 装配（NewCMMath GlyphAssembly），
/// 闭合填充字形，出场用 fade_in（填充字形禁用 create）。
fn arrow_demo(theme: &Theme, fps: u32) -> ffanim::Result<Timeline> {
    use ffanim::typst::{ArrowKind, TypstEngine};

    // 括注为中文：须 CJK 字形（find_cjk_font 优先，缺失时回退系统拉丁字体）
    let font = ffanim::typst::find_cjk_font()
        .and_then(|(path, idx)| ffanim::Font::from_index(&path, idx).ok())
        .map(Ok)
        .unwrap_or_else(|| Font::from_system(None::<&str>))?;
    // 箭头 = Typst MATH stretch 装配，与公式排版同引擎
    let mut engine = TypstEngine::new(8);
    let accent = theme.accent().unwrap();
    let fg = theme.foreground().unwrap();
    let (w, h) = (960u32, 540u32);
    let mut tl = Timeline::new(w, h, fps, theme.background().unwrap());

    // 1) 标题淡入
    let text_path = font.outline_text("Arrow 箭头", 56.0).unwrap();
    let bounds = text_path.bounding_box();
    let dx = f64::from(w) / 2.0 - (bounds.x0 + bounds.x1) / 2.0;
    let title = tl.add(
        Item::from_path(text_path)
            .with_fill(fg)
            .with_transform(Affine::translate((dx, f64::from(h) * 0.10))),
    );
    tl.play(Anim::fade_in(title).during(0.7));

    // 2) 三支箭头依次淡入 + 括注（交替前景/强调色；括注在杆中点上方居中）
    let specs: [(ArrowKind, Point, Point, &str); 3] = [
        (
            ArrowKind::Single,
            Point::new(220.0, 300.0),
            Point::new(740.0, 300.0),
            "单头",
        ),
        (
            ArrowKind::Single,
            Point::new(280.0, 460.0),
            Point::new(680.0, 380.0),
            "斜向",
        ),
        (
            ArrowKind::Double,
            Point::new(300.0, 200.0),
            Point::new(660.0, 200.0),
            "双头",
        ),
    ];
    for (i, (kind, from, to, label)) in specs.into_iter().enumerate() {
        let arrow = engine.arrow(from, to, kind)?;
        // 括注：杆中点上方 34px，bbox 中心对齐杆中点
        let mid = Point::new((from.x + to.x) / 2.0, (from.y + to.y) / 2.0);
        let label_path = font.outline_text(label, 30.0).unwrap();
        let lb = label_path.bounding_box();
        let lx = mid.x - (lb.x0 + lb.x1) / 2.0;
        let ly = mid.y - 34.0 - (lb.y0 + lb.y1) / 2.0;
        let color = if i % 2 == 0 { fg } else { accent };
        let (a, t) = (
            tl.add(arrow.with_fill(color)),
            tl.add(
                Item::from_path(label_path)
                    .with_fill(color)
                    .with_transform(Affine::translate((lx, ly))),
            ),
        );
        tl.wait(0.15);
        tl.together(
            [Anim::fade_in(a).during(0.8)]
                .into_iter()
                .chain([Anim::fade_in(t).during(0.6)]),
        );
    }

    tl.wait(0.8);
    Ok(tl)
}

/// M9.15 验收场景：P1-6 数字滚动（manim `DecimalNumber` + ValueTracker 对应）。
///
/// 标题淡入 → 主数字 0→1,234.5 千分位滚动（中心锚定不漂移）→ 副数字
/// 带符号 -40→+40 滚动。数字 = Typst markup 逐帧排版（闭合填充字形，
/// 出场 fade_in 禁 create）；tracker 值动画驱动，updater 每帧重建形状。
fn decimal_demo(theme: &Theme, fps: u32) -> ffanim::Result<Timeline> {
    use ffanim::typst::{DecimalNumberOptions, TypstEngine};

    // 括注为中文：须 CJK 字形（find_cjk_font 优先，缺失时回退系统拉丁字体）
    let font = ffanim::typst::find_cjk_font()
        .and_then(|(path, idx)| ffanim::Font::from_index(&path, idx).ok())
        .map(Ok)
        .unwrap_or_else(|| Font::from_system(None::<&str>))?;
    let fg = theme.foreground().unwrap();
    let accent = theme.accent().unwrap();
    let (w, h) = (960u32, 540u32);
    let mut tl = Timeline::new(w, h, fps, theme.background().unwrap());

    // 1) 标题淡入
    let text_path = font.outline_text("Decimal 数字滚动", 56.0).unwrap();
    let bounds = text_path.bounding_box();
    let dx = f64::from(w) / 2.0 - (bounds.x0 + bounds.x1) / 2.0;
    let title = tl.add(
        Item::from_path(text_path)
            .with_fill(fg)
            .with_transform(Affine::translate((dx, f64::from(h) * 0.10))),
    );
    tl.play(Anim::fade_in(title).during(0.7));

    // 2) 主数字：0 → 1,234.5 千分位滚动（中心锚定，updater 每帧重排）
    let anchor = Point::new(f64::from(w) / 2.0, 250.0);
    let sig = TypstEngine::decimal_signal(
        anchor,
        DecimalNumberOptions {
            decimals: 1,
            group_with_commas: true,
            include_sign: false,
            font_size_pt: 36.0,
            text_color: fg,
        },
    )?;
    let t = tl.add_tracker(0.0);
    let num = tl.add_updater(&[t], sig);
    tl.play(Anim::fade_in(num).during(0.6));
    tl.play(Anim::set_value(t, 1234.5).during(2.5));
    tl.wait(0.5);

    // 3) 副数字：-40 → +40 带符号滚动（include_sign：正数显 +）
    let sig2 = TypstEngine::decimal_signal(
        Point::new(f64::from(w) / 2.0, 380.0),
        DecimalNumberOptions {
            decimals: 0,
            include_sign: true,
            group_with_commas: false,
            font_size_pt: 28.0,
            text_color: accent,
        },
    )?;
    let t2 = tl.add_tracker(-40.0);
    let num2 = tl.add_updater(&[t2], sig2);
    tl.play(Anim::fade_in(num2).during(0.6));
    tl.play(Anim::set_value(t2, 40.0).during(2.0));
    tl.wait(0.6);
    Ok(tl)
}

/// M2 验收场景：三角形描边生长 → 圆形滑入 → 圆形形变放大 → 文字淡入。
fn demo_anim(theme: &Theme, font: &Font, fps: u32) -> Timeline {
    let accent = theme.accent().unwrap();
    let foreground = theme.foreground().unwrap();
    let sw = f64::from(theme.style.stroke_width);
    let (w, h) = (960u32, 540u32);

    let mut tl = Timeline::new(w, h, fps, theme.background().unwrap());

    // 三角形：左侧，描边生长
    let tri = Item::regular_polygon((f64::from(w) * 0.25, f64::from(h) * 0.62).into(), 120.0, 3)
        .with_stroke(accent)
        .with_stroke_width(sw);
    let tri = tl.add(tri);
    tl.play(Anim::create(tri).during(1.2));

    // 圆形：右侧，从下方滑入淡入

/// M4 验收场景：坐标系 + 正弦函数图像随 ValueTracker 动画（求值期信号）。

fn sine_demo(theme: &Theme, fps: u32) -> ffanim::Result<Timeline> {

/// M9.11 验收场景（P1-5）：TransformMatchingShapes 任意片段结构匹配形变。
///
/// 标题淡入 → "MATH" 逐字形淡入 → `match_shapes` 几何签名配对 →
/// M/A/T 跨行形变（T 异位滑动）、H 淡出、S 淡入。
fn shapes_demo(theme: &Theme, fps: u32) -> ffanim::Result<Timeline> {
    use ffanim::{Color, ItemId};

    let font = Font::from_system(None::<&str>)?;
    let accent = theme.accent().unwrap();
    let fg = theme.foreground().unwrap();
    let (w, h) = (960u32, 540u32);
    let mut tl = Timeline::new(w, h, fps, theme.background().unwrap());

    // 1) 标题淡入
    let text_path = font.outline_text("TransformMatchingShapes", 52.0).unwrap();
    let bounds = text_path.bounding_box();
    let dx = f64::from(w) / 2.0 - (bounds.x0 + bounds.x1) / 2.0;
    let title = tl.add(
        Item::from_path(text_path)
            .with_fill(fg)
            .with_transform(Affine::translate((dx, f64::from(h) * 0.12))),
    );
    tl.play(Anim::fade_in(title).during(0.7));

    // 逐字形散件：单字符 outline_text（同字形 base path 相同，位置烘焙在坐标里）
    let glyphs = |word: &str, size: f64, cx: f64, baseline: f64, color: Color| -> Vec<Item> {
        let adv = size * 0.78;
        let n = word.chars().count() as f64;
        word.chars()
            .enumerate()
            .map(|(i, c)| {
                let path = font.outline_text(&c.to_string(), size).unwrap();
                let x = cx - adv * (n - 1.0) / 2.0 + i as f64 * adv;
                Item::from_path(path)
                    .with_fill(color)
                    .with_transform(Affine::translate((x, baseline)))
            })
            .collect()
    };

    // 2) "MATH" 逐字形淡入
    let a_parts = glyphs("MATH", 96.0, 480.0, 250.0, accent);
    let ids_a: Vec<ItemId> = a_parts.iter().map(|it| tl.add(it.clone())).collect();
    tl.group(ids_a.iter().map(|&id| Anim::fade_in(id).during(0.8)), 0.08);
    tl.wait(0.4);

    // 3) match_shapes 配对 → M/A/T 形变（T 异位滑动）、H 淡出、S 淡入
    let b_parts = glyphs("MAST", 96.0, 480.0, 420.0, fg);
    let ids_b: Vec<ItemId> = b_parts.iter().map(|it| tl.add(it.clone())).collect();
    let plan = ffanim::match_shapes(&a_parts, &b_parts);
    let dur = 1.4;
    let anims = plan
        .paired
        .iter()
        .map(|&(i, j)| Anim::transform(ids_a[i], ids_b[j]).during(dur))
        .chain(
            plan.leave_a
                .iter()
                .map(|&k| Anim::fade_out(ids_a[k]).during(dur)),
        )
        .chain(
            plan.enter_b
                .iter()
                .map(|&k| Anim::fade_in(ids_b[k]).during(dur)),
        );
    tl.together(anims);
    tl.wait(0.8);
    Ok(tl)
}

//! 产物：out/fourier.mp4 + strip/onion 诊断 PNG。
//! 配音：`BGM=路径.wav cargo run ...` 自动混流并延伸片长（FromAudio 语义）。
//! 示例 BGM：Erik Satie《Gymnopédie No.1》开头主题（Wikimedia Commons，
//! https://upload.wikimedia.org/wikipedia/commons/9/90/Erik_Satie_-_gymnopedies_-_la_1_ere._lent_et_douloureux.ogg
//! 许可：Public domain，可自由分发）——裁剪 18s + 响度归一 -18 LUFS + 淡入淡出。

use ffanim::kurbo::{BezPath, Vec2};
use ffanim::typst::{FormulaOptions, TextOptions, TypstEngine};
use ffanim::{Anim, Axes, Camera, Color, Font, Item, RateFunc, Theme, Timeline};

const W: u32 = 960;
const H: u32 = 540;
const FPS: u32 = 30;

fn main() -> ffanim::Result<()> {
    // 数据驱动：颜色与描边宽度来自主题 JSON（仓库根 themes/）
    let theme_path = concat!(env!("CARGO_MANIFEST_DIR"), "/../../themes/default.json");
    let theme = Theme::from_path(theme_path)?;
    let accent = theme.accent()?;
    let fg = theme.foreground()?;
    let mut tl = Timeline::new(W, H, FPS, theme.background()?);

    // 排版引擎 + 系统中文字体（env FFANIM_CJK_FONT → fc-list :lang=zh → 常见路径，
    // 无硬编码路径；缺失时 Typst 退回内置拉丁字体）
    let mut engine = TypstEngine::new(16);
    if let Some((font, idx)) = ffanim::typst::find_cjk_font() {
        engine.add_font_face(&font, idx)?;
    }
    // strip 时间轴标注用拉丁字体（与文本排版无关）
    let font = Font::from_system(None::<&str>)?;

    // —— 幕 1：标题淡入 ——
    let title = engine.text("傅里叶级数与方波", &TextOptions::new(52.0, fg))?;
    let title_place = title.fit_transform(f64::from(W) / 2.0, 48.0 + title.height_pt / 2.0, 1.0);
    let title_ids: Vec<_> = title
        .to_items()
        .into_iter()
        .map(|item| tl.add(item.with_transform(title_place)))
        .collect();
    tl.group(
        title_ids.iter().map(|&id| Anim::fade_in(id).during(1.0)),
        0.0,
    );
    tl.wait(0.3);

    // —— 幕 2：级数公式逐字形写出 ——
    let fopts = FormulaOptions {
        font_size_pt: 30.0,
        text_color: fg,
    };
    let formula = engine.formula(
        "f(x) = 4/pi (sin x + frac(sin 3x, 3) + frac(sin 5x, 5) + dots)",
        &fopts,
    )?;
    // 缩放后居中于标题下方（fit_transform 修正 center_transform 的缩放偏移）
    let place = formula.fit_transform(f64::from(W) / 2.0, 158.0, 1.5);
    let write_dur = 0.5;
    let step = 0.06;
    // 公式逐字写出：group 按步长错开排入（M8.3 组动画，替代逐段 play+overlap 手工对齐）
    let ids: Vec<_> = formula
        .to_items()
        .into_iter()
        .map(|piece| tl.add(piece.with_transform(place)))
        .collect();
    tl.group(
        ids.iter()
            .map(|&id| Anim::create(id).during(write_dur).rate(RateFunc::Smooth)),
        step,
    );
    tl.wait(0.5);

    // —— 幕 3：坐标系 + 方波目标 ——
    let axes = Axes::new((-3.5, 3.5), (-1.5, 1.5))
        .with_size(880.0, 270.0)
        .with_center(f64::from(W) / 2.0, 350.0);
    let grid = tl.add(axes.grid_item(fg.with_alpha(40), 1.0));
    tl.play(Anim::fade_in(grid).during(0.6));
    let ax = tl.add(axes.to_item(&theme.style));
    tl.overlap(0.4);
    tl.play(Anim::create(ax).during(1.0));
    // 方波目标：sign(sin(πx/2))，细线淡入（半透明前景色）
    let square = tl.add(square_wave_item(&axes, fg.with_alpha(130)));
    tl.overlap(0.6);
    tl.play(Anim::fade_in(square).during(0.8));
    tl.wait(0.3);

    // —— 幕 4：部分和 S_N 随 tracker N 逐阶升高 ——
    let sum = tl.add_tracker(1.0);
    let curve = tl.add_updater(&[sum], move |v| {
        let nn = v[0].round() as i64;
        axes.plot(
            |x| {
                // S_N(x) = 4/π · Σ_{k=1,3,..,N} sin(kπx/2)/k
                let mut s = 0.0;
                let mut k = 1;
                while k <= nn {
                    s += (k as f64 * x * std::f64::consts::FRAC_PI_2).sin() / k as f64;
                    k += 2;
                }
                4.0 / std::f64::consts::PI * s
            },
            (-3.5, 3.5),
        )
        .with_stroke(accent)
        .with_stroke_width(3.0)
    });
    tl.play(Anim::create(curve).during(1.5));
    tl.wait(0.3);
    // 逐阶升高：N = 1 → 3 → 5 → 7 → 9（跳变处过冲随之成形，吉布斯现象）
    for target in [3.0, 5.0, 7.0, 9.0] {
        tl.play(Anim::set_value(sum, target).during(1.2));
        tl.wait(0.2);
    }

    // —— 推近镜头（M8.1 相机系统验收段）：相机即 Item，transform_to 到目标仿射 ——
    let camera = tl.camera(Camera::at(W, H));
    let push_target = Camera {
        center: Vec2::new(f64::from(W) / 2.0, 350.0), // 注视方波曲线中心
        zoom: 1.4,
        rotation: 0.0,
    };
    tl.play(
        Anim::transform_to(camera, push_target.affine(W, H))
            .during(1.4)
            .rate(RateFunc::Smooth),
    );

    // —— 幕 5：结论 ——
    let caption = engine.text(
        "N → ∞ 时逼近方波，跳变处残留 ≈ 9% 过冲（吉布斯现象）",
        &TextOptions::new(26.0, fg),
    )?;
    let cap_place = caption.fit_transform(f64::from(W) / 2.0, 495.0 + caption.height_pt / 2.0, 1.0);
    let cap_ids: Vec<_> = caption
        .to_items()
        .into_iter()
        .map(|item| tl.add(item.with_transform(cap_place)))
        .collect();
    tl.group(cap_ids.iter().map(|&id| Anim::fade_in(id).during(0.8)), 0.0);
    tl.wait(1.2);

    let mut tl = tl.compile();

    // —— 出片 + 无看片诊断（agent 验证循环的库式用法）——
    // BGM 环境变量指向音频文件时：analyze → 片长延伸到音频长度（FromAudio 语义）
    // → 渲染临时文件 → 混流为成品
    let bgm_path = std::env::var("BGM").ok();
    let encode_out: std::path::PathBuf = match &bgm_path {
        Some(bgm) => {
            let report = ffanim::audio::analyze(std::path::Path::new(bgm))?;
            let frames = (report.duration * f64::from(FPS)).ceil() as usize;
            println!(
                "BGM {bgm}: {:.2}s（RMS {:.1}dBFS）→ 片长延伸到 {frames} 帧",
                report.duration, report.rms_db
            );
            tl = tl.with_min_duration(frames);
            "out/fourier_tmp.mp4".into()
        }
        None => "out/fourier.mp4".into(),
    };
    ffanim::render_to_mp4(&tl, &encode_out)?;
    println!(
        "已输出 {encode_out:?}（{:.2}s）",
        tl.duration_frames as f64 / f64::from(FPS)
    );
    if let Some(bgm) = &bgm_path {
        ffanim::audio::mux(
            std::path::Path::new(&encode_out),
            std::path::Path::new(bgm),
            std::path::Path::new("out/fourier.mp4"),
        )?;
        std::fs::remove_file(&encode_out).ok();
        println!("音轨已混入 out/fourier.mp4");
    }

    let report = ffanim::tools::strip(
        &tl,
        Some(&font),
        accent,
        "out/fourier_strip.png".into(),
        12,
        4,
        320,
    )
    .map_err(ffanim::Error::Other)?;
    println!("strip: {:?}（采样帧 {:?}）", report.out, report.frames);

    // onion 取部分和阶跃段中点（N: 3→5 期间）
    let onion_frame = tl.duration_frames * 2 / 3;
    let report = ffanim::tools::onion(&tl, onion_frame, 6, "out/fourier_onion.png".into())
        .map_err(ffanim::Error::Other)?;
    println!("onion: 基准帧 {} → {:?}", report.frame, report.out);

    Ok(())
}

/// 方波目标曲线：sign(sin(πx/2))，断点 x = -2, 0, 2（数据坐标 → 像素）。
fn square_wave_item(axes: &Axes, color: Color) -> Item {
    let mut path = BezPath::new();
    let pts = [
        (-3.5, -1.0),
        (0.0, -1.0),
        (0.0, 1.0),
        (2.0, 1.0),
        (2.0, -1.0),
        (3.5, -1.0),
    ];
    for (i, (x, y)) in pts.iter().enumerate() {
        let p = axes.c2p(*x, *y);
        if i == 0 {
            path.move_to(p);
        } else {
            path.line_to(p);
        }
    }
    Item::from_path(path)
        .with_stroke(color)
        .with_stroke_width(2.0)
}

//! M8.4 验收示例：SVG 导入淡入 + 位图图片滑入缩放。
//!
//! 运行：`cargo run -p ffanim --example media`，产物 `out/media.mp4`。

use ffanim::kurbo::Affine;
use ffanim::{Anim, Color, Item, Timeline, from_svg_str};

const W: f64 = 960.0;
const H: f64 = 540.0;

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let bg = Color::new(14, 17, 22, 255);
    let mut tl = Timeline::new(W as u32, H as u32, 30, bg);

    // 1. SVG：三色徽标（两圆一矩形），整组淡入
    let svg = r##"<svg xmlns="http://www.w3.org/2000/svg" width="120" height="120">
        <circle cx="40" cy="60" r="28" fill="#e05d44"/>
        <circle cx="80" cy="60" r="28" fill="#f2b134"/>
        <rect x="52" y="52" width="16" height="16" fill="#3b88c3"/>
    </svg>"##;
    let logo = tl.add(from_svg_str(svg)?.with_translate(W / 2.0 - 60.0, 90.0));
    tl.play(Anim::fade_in(logo).during(1.0));

    // 2. 位图：程序生成 64x64 渐变方块，淡入后滑向左下并放大 3 倍
    let mut rgba = vec![0u8; 64 * 64 * 4];
    for y in 0..64u32 {
        for x in 0..64u32 {
            let i = ((y * 64 + x) * 4) as usize;
            rgba[i] = (x * 4) as u8;
            rgba[i + 1] = (y * 4) as u8;
            rgba[i + 2] = 180;
            rgba[i + 3] = 255;
        }
    }
    let img = tl.add(Item::from_image(rgba, 64, 64)?.with_translate(W / 2.0 - 32.0, H - 200.0));
    tl.play(Anim::fade_in(img).during(1.0));
    tl.play(
        Anim::transform_to(
            img,
            Affine::translate((W / 2.0 - 96.0, H - 264.0)) * Affine::scale(3.0),
        )
        .during(1.0),
    );

    let ct = tl.compile();
    ffanim::render_to_mp4(&ct, "out/media.mp4")?;
    println!("OK: out/media.mp4（{} 帧）", ct.duration_frames);
    Ok(())
}

/// M9.18 验收场景：P1-14 表格（manim `Table` 对应）。
///
/// 标题淡入 → 表格行组 `lagged_start` 逐行淡入 → 第二行高亮底板淡入
/// → `surrounding_rect` 圈注「示例」格（create 描边生长）。
/// 表格 = core 侧逐格排版拼组（行组保留独立动画句柄）；z 序 = 注册序，
/// 高亮底板须先于行组注册才能垫底。
fn table_demo(theme: &Theme, fps: u32) -> ffanim::Result<Timeline> {
    let fg = theme.foreground().unwrap();
    let accent = theme.accent().unwrap();
    let (w, h) = (960u32, 540u32);
    let mut tl = Timeline::new(w, h, fps, theme.background().unwrap());

    // 表格文本/标题须 CJK 字形（find_cjk_font 优先，缺失时回退拉丁字体）
    let font = ffanim::typst::find_cjk_font()
        .and_then(|(p, idx)| ffanim::Font::from_index(&p, idx).ok())
        .map(Ok)
        .unwrap_or_else(|| Font::from_system(None::<&str>))?;

    // 1) 标题淡入（描边路径文本）
    let title_path = font.outline_text("复杂度对照表", 52.0)?;
    let bounds = title_path.bounding_box();
    let title = tl.add(Item::from_path(title_path).with_fill(fg).with_transform(
        Affine::translate((
            f64::from(w) / 2.0 - (bounds.x0 + bounds.x1) / 2.0,
            f64::from(h) * 0.10,
        )),
    ));
    tl.play(Anim::fade_in(title).during(0.7));
    tl.wait(0.2);

    // 2) 表格拼装（数据驱动：色皆出自主题，仅透明度派生）
    let o = ffanim::TableOptions {
        header_fill: Some(ffanim::Color::new(accent.r, accent.g, accent.b, 60)),
        ..ffanim::TableOptions::new(24.0, fg)
    };
    let spec = ffanim::table(
        &["运算", "复杂度", "示例"],
        &[
            vec!["求和", "O(n)", "sum(xs)"],
            vec!["排序", "O(n log n)", "sorted(xs)"],
        ],
        &font,
        &o,
    )?;
    // 表格从原点生长 → 平移到画布水平居中
    let b = spec.table.bbox();
    let move_t = Affine::translate((
        f64::from(w) / 2.0 - (b.x0 + b.x1) / 2.0,
        310.0 - (b.y0 + b.y1) / 2.0,
    ));

    // 高亮底板：先注册（z 序垫底），后播放
    let hl = tl.add(
        Item::rect(spec.row_rect(1))
            .with_transform(move_t)
            .with_fill(ffanim::Color::new(accent.r, accent.g, accent.b, 60)),
    );
    // 行组逐行注册 + 错峰淡入
    let row_ids: Vec<_> = spec
        .rows
        .iter()
        .map(|it| tl.add(it.clone().with_transform(move_t)))
        .collect();
    tl.lagged_start(row_ids.into_iter().map(Anim::fade_in), 0.5);
    tl.play(Anim::fade_in(hl).during(0.6));
    tl.wait(0.3);

    // 3) 圈注「示例」格：surrounding_rect 描边生长（未注册临时 Item 供定位）
    let ring = tl.add(
        Item::surrounding_rect(&Item::rect(spec.cell(1, 2)), 6.0)
            .with_transform(move_t)
            .with_stroke(accent),
    );
    tl.play(Anim::create(ring).during(0.6));
    tl.wait(0.8);
    Ok(tl)
}

/// M3 验收场景：公式逐字形写出（Write）→ 结构匹配形变（TransformMatchingTex）。
///
/// `x^2 + y^2 = z^2` 逐字形写出后，`+`/`z^2` 淡出、`-`/`0` 淡入，
/// 相同子式（x、2、y、=）跨公式逐点形变。
fn formula_demo(theme: &Theme, fps: u32) -> ffanim::Result<Timeline> {
    use ffanim::typst::FormulaOptions;
    use ffanim::{ItemId, RateFunc};

    let mut engine = ffanim::typst::TypstEngine::new(16);
    let fopts = FormulaOptions {
        font_size_pt: 32.0,
        text_color: theme.foreground().unwrap(),
    };
    let f1 = engine.formula("x^2 + y^2 = z^2", &fopts)?;
    let f2 = engine.formula("x^2 - y^2 = 0", &fopts)?;

    let (w, h) = (960.0_f64, 540.0_f64);
    let scale = 2.0; // pt → px
    let mut tl = Timeline::new(w as u32, h as u32, fps, theme.background().unwrap());
    // 公式局部坐标 → 画布居中（fit_transform：缩放后的中心精确落在目标点）
    let place = |f: &ffanim::typst::Formula| f.fit_transform(w / 2.0, h / 2.0, scale);

    // —— Write：逐字形 Create，group 按步长错开排入（M8.3 组动画，替代 overlap 手工回退）——
    let write_dur = 0.5;
    let step = 0.06;
    let ids_a: Vec<ItemId> = f1
        .to_items()
        .into_iter()
        .map(|item| tl.add(item.with_transform(place(&f1))))
        .collect();
    tl.group(
        ids_a
            .iter()
            .map(|&id| Anim::create(id).during(write_dur).rate(RateFunc::Smooth)),
        step,
    );
    tl.wait(0.6);

    // —— TransformMatchingTex：配对形变，其余淡出/淡入（同一窗口并行）——
    let plan = ffanim::typst::match_formulas(&f1, &f2);
    let mut ids_b: Vec<ItemId> = Vec::new();
    for item in f2.to_items() {
        ids_b.push(tl.add(item.with_transform(place(&f2))));
    }

    let dur = 1.2;
    let mut anims: Vec<Anim> = Vec::new();
    for (i, j) in &plan.paired {
        anims.push(Anim::transform(ids_a[*i], ids_b[*j]).during(dur));
    }
    // 淡出/淡入与形变同时长，保证所有动画共享同一窗口（overlap 回退对齐）
    for &k in &plan.leave_a {
        anims.push(Anim::fade_out(ids_a[k]).during(dur));
    }
    for &k in &plan.enter_b {
        anims.push(Anim::fade_in(ids_b[k]).during(dur));
    }
    for (k, anim) in anims.into_iter().enumerate() {
        if k > 0 {
            // 回退自身 during → 与首个动画同窗口起点（并行形变）
            tl.overlap(anim.duration_s);
        }
        tl.play(anim);
    }
    tl.wait(0.8);
    Ok(tl)
}

/// M8.2 验收场景：Text 富文本——中英混排限宽不出框 + 粗细混排 + 行距。
///
/// 系统中文字体经 `find_cjk_font` 数据查找注入引擎（无硬编码路径），

fn text_demo(theme: &Theme, fps: u32) -> ffanim::Result<Timeline> {

/// manim 对齐 P1-7 验收场景：标注复合件（圈选框/下划线/叉/背景板）。
///
/// 四个标注几何构造器均基于目标 bbox（含 transform）计算；标注基于注册基线，
/// 目标有位姿动画时需与标注同步挂相同位姿（第 1 段演示）。
fn annotate_demo(theme: &Theme, fps: u32) -> ffanim::Result<Timeline> {
    use ffanim::typst::{FormulaOptions, TextOptions, TypstEngine, find_cjk_font};

    let fg = theme.foreground().unwrap();
    let accent = theme.accent().unwrap();
    let sw = f64::from(theme.style.stroke_width);
    let (w, h) = (960u32, 540u32);
    let mut tl = Timeline::new(w, h, fps, theme.background().unwrap());

    let mut engine = TypstEngine::new(16);
    if let Some((font, idx)) = find_cjk_font() {
        engine.add_font_face(&font, idx)?;
    }
    let fopts = FormulaOptions {
        font_size_pt: 26.0,
        text_color: fg,
    };
    let topt = |size: f64, color: ffanim::Color| TextOptions::new(size, color);

    // 排版产物注册 helper：散件注册（返回句柄，供字形级动画）+ 拼组（仅几何用途，不注册）
    let reg = |tl: &mut Timeline, f: &ffanim::typst::Formula, place: Affine| -> (Vec<_>, Item) {
        let pieces: Vec<Item> = f
            .to_items()
            .into_iter()
            .map(|it| it.with_transform(place))
            .collect();
        let grp = Item::group(pieces.clone());
        (tl.add_group(pieces), grp)
    };

    // 1) 公式圈选：公式淡入 → surrounding_rect 描边生长 → 公式+框同步右移
    //    （标注基于注册基线，目标平移则标注挂相同 shift_to）
    let f = engine.formula("e^{i pi} + 1 = 0", &fopts)?;
    let (f_ids, f_grp) = reg(&mut tl, &f, f.fit_transform(f64::from(w) / 2.0, 120.0, 2.0));
    tl.group(f_ids.iter().map(|&id| Anim::fade_in(id).during(0.8)), 0.0);
    tl.wait(0.3);
    let frame = tl.add(
        Item::surrounding_rect(&f_grp, 18.0)
            .with_stroke(accent)
            .with_stroke_width(sw),
    );
    tl.play(Anim::create(frame).during(0.7));
    tl.wait(0.3);
    let mut sync = vec![Anim::shift_to(frame, 180.0, 0.0).during(0.8)];
    sync.extend(
        f_ids
            .iter()
            .map(|&id| Anim::shift_to(id, 180.0, 0.0).during(0.8)),
    );
    tl.together(sync);

    // 2) 下划线：单词淡入 → 底线描边
    let word = engine.text("underlined", &topt(24.0, fg))?;
    let (w_ids, w_grp) = reg(&mut tl, &word, word.fit_transform(240.0, 300.0, 2.0));
    tl.group(w_ids.iter().map(|&id| Anim::fade_in(id).during(0.6)), 0.0);
    tl.wait(0.2);
    let ul = tl.add(
        Item::underline(&w_grp, 10.0)
            .with_stroke(accent)
            .with_stroke_width(sw),
    );
    tl.play(Anim::create(ul).during(0.5));
    tl.wait(0.2);

    // 3) 叉：错误式子淡入 → 对角双线划掉
    let wrong = engine.text("2 + 2 = 5", &topt(24.0, fg))?;
    let (c_ids, c_grp) = reg(&mut tl, &wrong, wrong.fit_transform(700.0, 300.0, 2.0));
    tl.group(c_ids.iter().map(|&id| Anim::fade_in(id).during(0.6)), 0.0);
    tl.wait(0.3);
    let cross = tl.add(
        Item::cross(&c_grp, 12.0)
            .with_stroke(accent)
            .with_stroke_width(sw),
    );
    tl.play(Anim::create(cross).during(0.5));
    tl.wait(0.2);

    // 4) 背景板：先注册背景矩形（画在下）再注册文字（画在上），同窗淡入
    let cap = engine.text("背景板", &topt(28.0, fg))?;
    let cap_place = cap.fit_transform(f64::from(w) / 2.0, 450.0, 2.0);
    let cap_grp = Item::group(
        cap.to_items()
            .into_iter()
            .map(|it| it.with_transform(cap_place))
            .collect(),
    );
    let bg =
        tl.add(Item::background_rect(&cap_grp, 16.0).with_fill(ffanim::Color { a: 90, ..accent }));
    let cap_ids = tl.add_group(
        cap.to_items()
            .into_iter()
            .map(|it| it.with_transform(cap_place))
            .collect(),
    );
    tl.together(
        [Anim::fade_in(bg).during(0.7)]
            .into_iter()
            .chain(cap_ids.iter().map(|&id| Anim::fade_in(id).during(0.7))),
    );
    tl.wait(0.5);
    Ok(tl)
}

/// M9.9 验收场景（P1-13）：代码块语法高亮。

/// 标题淡入 → Rust 代码块逐字形写出（Dark 主题）→ Python 代码块淡入，

fn code_demo(theme: &Theme, fps: u32) -> ffanim::Result<Timeline> {
    use ffanim::typst::{CodeOptions, TextOptions, TypstEngine, find_cjk_font};

    let fg = theme.foreground().unwrap();
    let accent = theme.accent().unwrap();
    let sw = f64::from(theme.style.stroke_width);

    Ok(tl)

/// M9.16 验收场景：P1-12 按词着色（manim `Text(t2c=…)` 对应）。
